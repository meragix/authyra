## Flutter App

### 1. Install the package

```yaml
dependencies:
  flutter_authyra: ^0.1.0
```

### 2. Configure Authyra in main.dart

```dart [main.dart]
import 'package:flutter/material.dart';
import 'package:flutter_authyra/flutter_authyra.dart';

void main() async {
  WidgetsFlutterBinding.ensureInitialized();
  
  // Initialize client (restore session if exists)
  Authyra.instance.initialize(
    client: AuthyraClient(
      providers: [
        CredentialsProvider(
          id: 'email-login',
          authorize: (email, password) async {
            // TODO: Call your backend API
            // For now, accept any credentials
            return AuthUser(
              id: 'user-${DateTime.now().millisecondsSinceEpoch}',
              email: email,
              name: email.split('@').first,
            );
          },
        ),
      ],
      storage: SecureAuthStorage(),
    ),
  );
  
  runApp(const MyApp());
}
```

### 3. Wrap your app with AuthyraScope

```dart [my_app.dart]
class MyApp extends StatelessWidget {
  const MyApp({super.key});
  
  @override
  Widget build(BuildContext context) {
    return AuthyraScope(
      child: MaterialApp(
        home: AuthGuard(
          fallback: const LoginScreen(),
          child: const HomeScreen(),
        ),
      ),
    );
  }
}
```

### 4. Create a login screen

```dart [login_screen.dart]
class LoginScreen extends StatefulWidget {
  const LoginScreen({super.key});

  @override
  State<LoginScreen> createState() => _LoginScreenState();
}

class _LoginScreenState extends State<LoginScreen> {
  final _emailController = TextEditingController();
  final _passwordController = TextEditingController();
  bool _isLoading = false;
  String? _error;

  Future<void> _signIn() async {
    setState(() {
      _isLoading = true;
      _error = null;
    });

    try {
      await Authyra.instance.signIn(
        provider: 'email-login',
        credentials: {
          'email': _emailController.text,
          'password': _passwordController.text,
        },
      );
      
      // No manual navigation needed!
      // AuthGuard will automatically show HomeScreen
      
    } catch (e) {
      setState(() => _error = e.toString());
    } finally {
      if (mounted) {
        setState(() => _isLoading = false);
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Login')),
      body: Padding(
        padding: const EdgeInsets.all(24),
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            TextField(
              controller: _emailController,
              decoration: const InputDecoration(
                labelText: 'Email',
                border: OutlineInputBorder(),
              ),
              keyboardType: TextInputType.emailAddress,
            ),
            const SizedBox(height: 16),
            TextField(
              controller: _passwordController,
              decoration: const InputDecoration(
                labelText: 'Password',
                border: OutlineInputBorder(),
              ),
              obscureText: true,
            ),
            const SizedBox(height: 24),
            if (_error != null)
              Padding(
                padding: const EdgeInsets.only(bottom: 16),
                child: Text(_error!, style: const TextStyle(color: Colors.red)),
              ),
            SizedBox(
              width: double.infinity,
              child: FilledButton(
                onPressed: _isLoading ? null : _signIn,
                child: _isLoading
                    ? const SizedBox(
                        height: 20,
                        width: 20,
                        child: CircularProgressIndicator(strokeWidth: 2),
                      )
                    : const Text('Sign In'),
              ),
            ),
          ],
        ),
      ),
    );
  }
  
  @override
  void dispose() {
    _emailController.dispose();
    _passwordController.dispose();
    super.dispose();
  }
}
```

### 5. Create a home screen

```dart [home_screen.dart]
class HomeScreen extends StatelessWidget {
  const HomeScreen({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: AuthBuilder(
          builder: (context, state) {
            return Text('Hello ${state.account?.displayName ?? ""}');
          },
        ),
        actions: [
          IconButton(
            icon: const Icon(Icons.logout),
            onPressed: () async {
              await context.authSignOut();
              // AuthGuard will automatically show LoginScreen
            },
          ),
        ],
      ),
      body: Center(
        child: AuthBuilder(
          builder: (context, state) {
            final account = state.account;
            if (account == null) return const SizedBox.shrink();
            
            return Column(
              mainAxisAlignment: MainAxisAlignment.center,
              children: [
                CircleAvatar(
                  radius: 50,
                  child: Text(
                    account.displayName[0].toUpperCase(),
                    style: const TextStyle(fontSize: 40),
                  ),
                ),
                const SizedBox(height: 16),
                Text(
                  account.displayName,
                  style: Theme.of(context).textTheme.headlineMedium,
                ),
                Text(
                  account.email,
                  style: Theme.of(context).textTheme.bodyLarge,
                ),
              ],
            );
          },
        ),
      ),
    );
  }
}
```

### 6. Run your app

```bash
flutter run
```

**That's it!** You now have:

- ✅ Login/logout functionality
- ✅ Automatic route protection
- ✅ Session persistence (survives app restart)
- ✅ Reactive UI updates
